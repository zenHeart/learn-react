🧩 核心思想图解
// Vue: 子组件通过 slot 传递数据 → 父组件通过 v-slot 接收
// React: 子组件调用父组件传入的函数 → 父组件控制渲染

// 共同点：数据来源是子组件，控制渲染是父组件
// 这就是「控制反转」— Inversion of Control
📦 Render Props 模式 推荐
// MouseTracker.jsx
class MouseTracker extends React.Component {
  state = { x: 0, y: 0 }

  handleMouseMove = (e) => {
    this.setState({ x: e.clientX, y: e.clientY })
  }

  render() {
    // 调用父组件传入的 render 函数，传入内部状态
    return (
      <div onMouseMove={this.handleMouseMove}>
        {this.props.render(this.state)}
      </div>
    )
  }
}

// 使用处
<MouseTracker render={({ x, y }) => (
  <div>鼠标位置：{x}, {y}</div>
)} />

🎒 HOC（Higher-Order Components）模式
// withMousePosition.js
function withMousePosition(WrappedComponent) {
  return function EnhancedComponent(props) {
    const [position, setPosition] = useState({ x: 0, y: 0 })

    useEffect(() => {
      const handleMove = (e) => setPosition({ x: e.clientX, y: e.clientY })
      window.addEventListener('mousemove', handleMove)
      return () => window.removeEventListener('mousemove', handleMove)
    }, [])

    // 将 position 注入到被包装组件的 props 中
    return <WrappedComponent {...props} mousePosition={position} />
  }
}

// 使用
const EnhancedStatus = withMousePosition(StatusDisplay)
<EnhancedStatus />

🪝 自定义 Hooks 模式 现代主流
// useMousePosition.js
function useMousePosition() {
  const [position, setPosition] = useState({ x: 0, y: 0 })

  useEffect(() => {
    const handleMove = (e) => {
      setPosition({ x: e.clientX, y: e.clientY })
    }
    window.addEventListener('mousemove', handleMove)
    return () => window.removeEventListener('mousemove', handleMove)
  }, [])

  return position
}

// 使用 — 组件只负责渲染，逻辑全部在 Hook 中
function StatusDisplay() {
  const { x, y } = useMousePosition()
  return <div>鼠标位置：{x}, {y}</div>
}

// MouseTracker.jsx
function MouseTracker({ children }) {
  const [position, setPosition] = useState({ x: 0, y: 0 })

  useEffect(() => {
    const handleMove = (e) => setPosition({ x: e.clientX, y: e.clientY })
    window.addEventListener('mousemove', handleMove)
    return () => window.removeEventListener('mousemove', handleMove)
  }, [])

  // 克隆 children 并注入 mousePosition prop
  const childrenWithProps = React.Children.map(children, (child) =>
    React.isValidElement(child)
      ? React.cloneElement(child, { mousePosition: position })
      : child
  )

  return <div>{childrenWithProps}</div>
}

// 使用 — 类似 Vue slot
<MouseTracker>
  <StatusDisplay />
</MouseTracker>

// StatusDisplay 直接通过 props 接收
<div>{props.mousePosition?.x}, {props.mousePosition?.y}</div>
