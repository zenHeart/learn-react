📝 代码示例

// 基础使用
import Joyride, { CallBackProps } from 'react-joyride';

const steps = [
  {
    'target': '.first-step',
    'content': '欢迎使用！',
    'disableBeacon': true,
  },
  {
    'target': '.second-step',
    'content': '查看数据统计',
  },
];

function App() {
  const [run, setRun] = useState(false);

  return (
    <Joyride
      steps={steps}
      run={run}
      continuous
      showSkipButton
    />
  );
}

import Driver from 'driver.js';
import 'driver.js/dist/driver.min.css';

const driver = new Driver({
  'animate': true,
  'opacity': 0.75,
  'padding': 5,
  'allowClose': true,
});

driver.defineSteps([
  {
    'element': '#driver-step-1',
    'popover': {
      'title': '智能提示',
      'description': '...',
    },
  },
]);

driver.start();
